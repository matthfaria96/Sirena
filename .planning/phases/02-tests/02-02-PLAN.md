```swift
struct Window { let id: CGWindowID?; let rect: CGRect }
struct RectCalculationParameters { let window: Window; let visibleFrameOfScreen: CGRect; let action: WindowAction; let lastAction: RectangleAction? }
struct RectResult { let rect: CGRect; let resultingAction: WindowAction?; let subAction: SubWindowAction? }
enum HalfSplitSide { case leading, trailing }
struct HalfSplitFrameCalculation {
  static func horizontalRect(in: CGRect, side: HalfSplitSide, fraction: Float) -> CGRect   // trailing => origin.x = maxX - width
  static func verticalRect(in: CGRect, side: HalfSplitSide, fraction: Float) -> CGRect     // leading => top (origin.y = maxY - height)
}
class WindowCalculationFactory { static let leftHalfCalculation, rightHalfCalculation, topHalfCalculation, bottomHalfCalculation, centerCalculation, maximizeCalculation, ... }
```
- `LeftRightHalfCalculation.calculateFractionalRect(_ params:, fraction: Float) -> RectResult` (side from params.action == .rightHalf). `calculateFirstRect` reads `ActiveSideSplitRatios.shared`/`Defaults.horizontalSplitRatio` (user prefs) — do NOT assert on it with fixed numbers; use calculateFractionalRect(fraction: 0.5) and HalfSplitFrameCalculation instead.
- `FirstThirdCalculation.calculateRect`: with `lastAction: nil` returns `orientationBasedRect` (landscape: width = floor(w/3), subAction .leftThird; portrait: height = floor(h/3) at top, .topThird). Same pattern in CenterThirdCalculation / LastThirdCalculation (read them).
- `MaximizeCalculation.calculateRect` returns visibleFrameOfScreen.
- `CenterCalculation.calculateRect`: centers window rect with `round`; if both dims exceed frame returns frame with resultingAction .maximize; if one exceeds, clamps that dimension.
- Note: tests run inside the Sirena host process and read the user's real UserDefaults — never write Defaults from tests.
